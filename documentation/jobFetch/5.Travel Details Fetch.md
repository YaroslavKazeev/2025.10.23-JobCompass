```mermaid
sequenceDiagram
    autonumber
    actor User
    %% File paths:
    %% JobCard: client/src/components/JobCard/JobCard.jsx
    %% OpenPositions: client/src/pages/openPositions/OpenPositions.jsx
    %% JobsContext: client/src/context/JobsContext.jsx
    %% useFetch: client/src/hooks/useFetch.js
    %% TravelRouter: server/src/routes/travel.js
    %% TravelController: server/src/controllers/travelController.js
    %% GoogleMapsAPI: server/src/services/googleMapsApi.js
    participant JobCard as Job Card<br/>JobCard.jsx
    participant OpenPositions as OpenPositions Page<br/>OpenPositions.jsx
    participant ErrorDisplay as Error Message Div<br/>(.error-message)
    participant JobsContext as JobsContext Provider<br/>JobsContext.jsx
    participant useFetch_2 as useFetch Hook<br/>(travel batch)<br/>useFetch.js
    participant FrontendNetwork as HTTP Network Layer
    participant TravelRouter as Travel Routes<br/>travel.js
    participant TravelController as Travel Controller<br/>travelController.js<br/>calculateBatchTravelTime()
    participant GoogleMapsAPI as Google Maps API<br/>googleMapsApi.js

    Note over User,GoogleMapsAPI: Phase 5: Travel Details Fetch
    activate JobsContext
    JobsContext->>JobsContext: handleJobFetchResults():
    JobsContext->>JobsContext: setAllJobs(data.result)
    JobsContext->>JobsContext: if (data.msg) setServerMessage(data.msg)
    JobsContext->>JobsContext: citiesToFetch = getCitiesToFetch(data.result)<br/>(from job.display_location)

    alt No cities to fetch
        Note over JobsContext: If citiesToFetch.length === 0,<br/>return early without API call
    else Cities exist
        JobsContext->>JobsContext: Build homeAddress from user profile<br/>{ street, house_number, city, country }
        JobsContext->>useFetch_2: performTravelFetch POST /api/travel/batch with body { homeAddress, workCities }

        activate useFetch_2
        useFetch_2->>useFetch_2: setError(null)
        useFetch_2->>useFetch_2: setIsLoading(true)

        JobsContext->>JobCard: isTravelLoading true -> show spinner

        activate JobCard
        JobCard->>User: Display small spinner in commute info
        deactivate JobCard

        useFetch_2->>FrontendNetwork: POST /api/travel/batch

        FrontendNetwork->>TravelRouter: HTTP POST /api/travel/batch
        TravelRouter->>TravelController: calculateBatchTravelTime(req, res)
        activate TravelController
        TravelController->>TravelController: formatAddress(homeAddress)
        TravelController->>TravelController: Filter workCities to strings only
        TravelController->>TravelController: regex match for same city or NL provinces
        alt Same-city/province match
            TravelController-->>TravelController: travel_time = 0; transfers = 0
        else Needs lookup
            TravelController->>GoogleMapsAPI: getTransitRouteSummary(origin, workCity, apiKey)
            activate GoogleMapsAPI
            alt Google Maps success
                GoogleMapsAPI-->>TravelController: avg travel_time (min) & least_transfers
            else Google Maps failure
                GoogleMapsAPI-->>TravelController: { workCity, error }
            end
            deactivate GoogleMapsAPI
        end

        TravelController->>TravelController: Aggregate per-city results (success + errors)
        TravelController-->>FrontendNetwork: HTTP 200 { success: true, result: { travelDetails } }
        deactivate TravelController

        FrontendNetwork->>useFetch_2: Response received
        useFetch_2->>useFetch_2: Parse JSON
        useFetch_2->>useFetch_2: setIsLoading(false)
        useFetch_2->>JobsContext: handleTravelFetchResults(data)
        deactivate useFetch_2
        JobsContext->>JobsContext: Merge travelDetails map<br/>{ workCity: { travel_time, least_transfers } }
        JobsContext->>JobCard: isTravelLoading false -> show travel data when available
        JobCard-->>User: Render travel time / transfers (if present)
    end

    alt Server error or network failure
        FrontendNetwork-->>useFetch_2: Network/500 error
        activate useFetch_2
        useFetch_2->>useFetch_2: setError(error.message); setIsLoading(false)
        useFetch_2-->>JobsContext: expose travelFetchError
        deactivate useFetch_2

        JobsContext-->>OpenPositions: travelFetchError available
        activate OpenPositions
        OpenPositions->>ErrorDisplay: AlertMessage shows error text
        ErrorDisplay-->>User: "Error loading jobs or commute info: [error]"
        deactivate OpenPositions
    end
```
